create table sales_order_23
  (
  orderno varchar(6)primary key,
  clientno varchar(6) references client_master_23,
  orderdate date not null,
  delyaddr varchar(20),
  salesmanno varchar(6) references salesman_master_23,
  delytype char(1),
  billyn char(1),
  delydate date,
  orderstatus varchar(10),
  constraint chk_orderno check (orderno like 'O%'),
  constraint chk_delytype check (delytype in('p','f')),
  constraint chk_delydate check (delydate>orderdate),
  constraint chk_orderstatus check (orderstatus in('in process','fulfilled','backorder','cancelled'))

);

SELECT P.DESCRIPTION,SUM(S.QTYDISP) AS TOTAL_QTY_SOLD FROM PRODUCT_MASTER_23 P, SALES_ORDER_DETAILS_23 S 
WHERE P.PRODUCTNO = S.PRODUCTNO
GROUP BY P.DESCRIPTION;

SELECT PRODUCTNO,
SUM(QTYDISP * PRODUCTRATE) AS VALUE_SOLD
FROM SALES_ORDER_DETAILS_23
GROUP BY PRODUCTNO;

SELECT O.CLIENTNO,
AVG(D.QTYDISP) AS AVG_QTY_SOLD
FROM SALES_ORDER_23 O, SALES_ORDER_DETAILS_23 D
WHERE O.ORDERNO = D.ORDERNO
GROUP BY O.CLIENTNO
HAVING SUM(D.QTYDISP * D.PRODUCTRATE) = 15000;

SELECT SUM(D.QTYDISP * D.PRODUCTRATE) AS TOTAL_BILLED_ORDERS
FROM SALES_ORDER_23 O, SALES_ORDER_DETAILS_23 D
WHERE O.ORDERNO = D.ORDERNO
AND O.BILLYN = 'Y'
AND TO_CHAR(O.ORDERDATE,'MON') = 'JUN';

SELECT P.PRODUCTNO,
  P.DESCRIPTION,
D.QTYORDERED - D.QTYDISP AS QTY_TO_DELIVER
FROM PRODUCT_MASTER_23 P,
     SALES_ORDER_DETAILS_23 D,
     SALES_ORDER_23 O
WHERE P.PRODUCTNO = D.PRODUCTNO
AND D.ORDERNO = O.ORDERNO
AND TO_CHAR(O.DELYDATE,'MM-YYYY') = TO_CHAR(SYSDATE,'MM-YYYY')
AND D.QTYORDERED > D.QTYDISP;

HAVING COUNT(D.ORDERNO) > 1;
